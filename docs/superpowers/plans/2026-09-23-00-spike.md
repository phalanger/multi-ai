```rust
use std::io::{IsTerminal, Read, Write};
use std::time::{SystemTime, UNIX_EPOCH};

fn main() {
    let argv: Vec<String> = std::env::args().skip(1).collect();
    let mut stdin = String::new();
    if !std::io::stdin().is_terminal() {
        let _ = std::io::stdin().read_to_string(&mut stdin);
    }
    let ts_ms = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|d| d.as_millis() as u64)
        .unwrap_or(0);
    let line = serde_json::json!({
        "ts_ms": ts_ms,
        "argv": argv,
        "stdin": stdin,
        "zellij_session": std::env::var("ZELLIJ_SESSION_NAME").ok(),
        "zellij_pane": std::env::var("ZELLIJ_PANE_ID").ok(),
        "cwd": std::env::current_dir().ok(),
    });
    let path = std::env::var("MAI_HOOKLOG").unwrap_or_else(|_| {
        let home = std::env::var("USERPROFILE")
            .or_else(|_| std::env::var("HOME"))
            .unwrap_or_else(|_| ".".into());
        format!("{home}/mai-hooklog.jsonl")
    });
    let mut f = std::fs::OpenOptions::new()
        .create(true)
        .append(true)
        .open(path)
        .expect("open log");
    writeln!(f, "{line}").expect("write log");
}
```

- [ ] **Step 2: 写 `src/main.rs`**

以下代码按写计划时的 russh 0.5x API 编写。如果编译报错，以 docs.rs 上
所装版本的 `russh::client` 文档为准修正（spike 代码，不进正式代码）。

```rust
use std::sync::Arc;

use axum::extract::ws::{Message, WebSocket, WebSocketUpgrade};
use axum::response::{Html, IntoResponse};
use axum::routing::get;
use russh::{ChannelMsg, client};

struct Trust;

impl client::Handler for Trust {
    type Error = russh::Error;

    async fn check_server_key(
        &mut self,
        _key: &russh::keys::ssh_key::PublicKey,
    ) -> Result<bool, Self::Error> {
        Ok(true) // spike only: no host key check
    }
}

#[derive(serde::Deserialize)]
struct Resize {
    cols: u32,
    rows: u32,
}

fn env(k: &str) -> anyhow::Result<String> {
    std::env::var(k).map_err(|_| anyhow::anyhow!("missing env {k}"))
}

async fn bridge(mut ws: WebSocket) -> anyhow::Result<()> {
    let host = env("MAI_HOST")?;
    let port: u16 = std::env::var("MAI_PORT")
        .ok()
        .and_then(|p| p.parse().ok())
        .unwrap_or(22);
    let user = env("MAI_USER")?;
    let cfg = Arc::new(client::Config::default());
    let mut s = client::connect(cfg, (host.as_str(), port), Trust).await?;
    let auth = if let Ok(path) = env("MAI_KEY") {
        let key = russh::keys::load_secret_key(path, None)?;
        let hash = s.best_supported_rsa_hash().await?.flatten();
        let key = russh::keys::PrivateKeyWithHashAlg::new(Arc::new(key), hash);
        s.authenticate_publickey(&user, key).await?
    } else {
        s.authenticate_password(&user, env("MAI_PASSWORD")?).await?
    };
    anyhow::ensure!(auth.success(), "auth failed");
    let mut ch = s.channel_open_session().await?;
    ch.request_pty(false, "xterm-256color", 120, 40, 0, 0, &[])
        .await?;
    ch.exec(true, env("MAI_CMD")?).await?;
    loop {
        tokio::select! {
            msg = ch.wait() => match msg {
                Some(ChannelMsg::Data { data })
                | Some(ChannelMsg::ExtendedData { data, .. }) => {
                    ws.send(Message::Binary(data.to_vec().into())).await?;
                }
                Some(ChannelMsg::ExitStatus { .. }) | None => break,
                _ => {}
            },
            m = ws.recv() => match m {
                Some(Ok(Message::Binary(b))) => ch.data(&b[..]).await?,
                Some(Ok(Message::Text(t))) => {
                    let r: Resize = serde_json::from_str(t.as_str())?;
                    ch.window_change(r.cols, r.rows, 0, 0).await?;
                }
                Some(Ok(Message::Close(_))) | None => break,
                _ => {}
            },
        }
    }
    Ok(())
}

async fn ws_handler(up: WebSocketUpgrade) -> impl IntoResponse {
    up.on_upgrade(|ws| async {
        if let Err(e) = bridge(ws).await {
            eprintln!("bridge error: {e:#}");
        }
    })
}

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    let app = axum::Router::new()
        .route("/", get(|| async { Html(include_str!("../static/index.html")) }))
        .route("/ws", get(ws_handler));
    let l = tokio::net::TcpListener::bind("127.0.0.1:7681").await?;
    println!("open http://127.0.0.1:7681/");
    axum::serve(l, app).await?;
    Ok(())
}
```
